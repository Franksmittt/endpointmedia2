function createEntry(
  path: string,
  changeFrequency: MetadataRoute.Sitemap[number]['changeFrequency'],
  priority: number,
  lastModified?: Date,
): MetadataRoute.Sitemap[number] {
  const cleanPath = path === '/' ? '' : path.replace(/\/$/, '');
  return {
    url: cleanPath === '' ? baseUrl : `${baseUrl}${cleanPath}`,
    lastModified: lastModified ?? new Date(),
    changeFrequency,
    priority,
  };
}

export function isDistinctEntity(content: {
  wordCount: number;
  uniqueFields: number;
  minWords?: number;
  minUnique?: number;
}): boolean {
  const minWords = content.minWords ?? 300;
  const minUnique = content.minUnique ?? 3;
  return content.wordCount >= minWords && content.uniqueFields >= minUnique;
}

import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';

const CRAWLER_UA =
  /Googlebot|Google-InspectionTool|AdsBot-Google|Mediapartners-Google|bingbot|Baiduspider|YandexBot|DuckDuckBot|Slurp|facebookexternalhit|Twitterbot|LinkedInBot|Applebot/i;

export function proxy(request: NextRequest) {
  const ua = request.headers.get('user-agent') ?? '';
  const isBot = CRAWLER_UA.test(ua);

  if (isBot) {
    const response = NextResponse.next();
    response.headers.set('x-robots-tag', 'all');
    return response;
  }

  return NextResponse.next();
}

      );
    if (isChunkError && typeof window !== 'undefined') {
      const reloadKey = 'chunk_reload_once';
      if (!sessionStorage.getItem(reloadKey)) {
        sessionStorage.setItem(reloadKey, '1');
        window.location.reload();
        return;
      }
    }
  }

  render() {
    if (this.state.hasError) {
      return (
        <div role="alert" className="p-8 text-center">
          <h1>Page temporarily unavailable</h1>
          <p>
            <a href="/">Return to homepage</a> or{' '}
            <button type="button" onClick={() => window.location.reload()}>
              reload
            </button>
          </p>
        </div>
      );
    }
    return this.props.children;
  }
}
